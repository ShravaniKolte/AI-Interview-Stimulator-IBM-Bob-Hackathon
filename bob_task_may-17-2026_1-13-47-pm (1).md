"""
Reset database script - Creates a fresh database with correct schema.
This will delete all existing data, so use with caution!
"""
import os
import shutil
from datetime import datetime

DB_PATH = "database/portfolio.db"

def reset_database():
    # Create backup if database exists
    if os.path.exists(DB_PATH):
        backup_path = f"database/portfolio_backup_{datetime.now().strftime('%Y%m%d_%H%M%S')}.db"
        print(f"[INFO] Creating backup at {backup_path}")
        shutil.copy2(DB_PATH, backup_path)
        print(f"[OK] Backup created successfully")
        
        # Delete old database
        os.remove(DB_PATH)
        print(f"[OK] Deleted old database")
    else:
        print("[INFO] No existing database found")
    
    # Import and create new database with correct schema
    print("[INFO] Creating new database with correct schema...")
    from database.db import engine
    from database import models
    
    models.Base.metadata.create_all(bind=engine)
    print("[OK] New database created successfully with all required columns")
    
    # Verify the schema
    import sqlite3
    conn = sqlite3.connect(DB_PATH)
    cursor = conn.cursor()
    cursor.execute("PRAGMA table_info(sessions)")
    columns = [col[1] for col in cursor.fetchall()]
    print(f"[INFO] Sessions table columns: {columns}")
    conn.close()
    
    print("[OK] Database reset complete!")

if __name__ == "__main__":
